% -- 1 --
\ExerciseSlide[\CategoryBadge[TrickColor!20]{Mno\. zenie przez 11}]{1}{30 s}{$23 \times 11 = ?$}
\AnswerSlide[\CategoryBadge[TrickColor!20]{Mno\. zenie przez 11}]{253}{Cyfry na zewn\k{a}trz, ich suma w \'srodek: $2\,|\,5\,|\,3$.}

% -- 2 --
\ExerciseSlide[\CategoryBadge[TrickColor!20]{Mno\. zenie przez 5}]{1}{30 s}{$48 \times 5 = ?$}
\AnswerSlide[\CategoryBadge[TrickColor!20]{Mno\. zenie przez 5}]{240}{Po\l{}owa razy 10: $24 \times 10$.}

% -- 3 --
\ExerciseSlide[\CategoryBadge[TrickColor!20]{Kwadrat na 5}]{2}{60 s}{$35^2 = ?$}
\AnswerSlide[\CategoryBadge[TrickColor!20]{Kwadrat na 5}]{1225}{$3 \times 4 = 12$, dopisz $25$.}

% -- 4 --
\ExerciseSlide[\CategoryBadge[TrickColor!20]{Procenty}]{2}{60 s}{$4\%$ z $75$}
\AnswerSlide[\CategoryBadge[TrickColor!20]{Procenty}]{3}{Procenty wolno odwr\'{o}ci\'c: $75\%$ z $4$.}

% -- 5 --
\ExerciseSlide[\CategoryBadge[TrickColor!20]{Mno\. zenie przez 9}]{2}{60 s}{$37 \times 9 = ?$}
\AnswerSlide[\CategoryBadge[TrickColor!20]{Mno\. zenie przez 9}]{333}{$370 - 37$.}

% -- 6 --
\ExerciseSlide[\CategoryBadge[TrickColor!20]{Po\l{}owa i podw\'{o}jnie}]{3}{90 s}{$16 \times 25 = ?$}
\AnswerSlide[\CategoryBadge[TrickColor!20]{Po\l{}owa i podw\'{o}jnie}]{400}{$4 \times 100$ --- dwa razy p\'{o}\l{} na jedno, dwa razy wi\k{e}cej na drugie.}

% -- 7 --
\ExerciseSlide[\CategoryBadge[TrickColor!20]{Mno\. zenie przez 11}]{3}{2 min}{%
    $67 \times 11 = ?$\par
    Uwaga: suma cyfr przekracza $9$.
  }
\AnswerSlide[\CategoryBadge[TrickColor!20]{Mno\. zenie przez 11}]{737}{$6+7=13$, wi\k{e}c jedynka idzie do przodu: $6{+}1\,|\,3\,|\,7$.}

% -- 8 --
\ExerciseSlide[\CategoryBadge[TrickColor!20]{R\'{o}\. znica kwadrat\'{o}w}]{3}{2 min}{$48 \times 52 = ?$}
\AnswerSlide[\CategoryBadge[TrickColor!20]{R\'{o}\. znica kwadrat\'{o}w}]{2496}{$50^2 - 2^2 = 2500 - 4$.}
