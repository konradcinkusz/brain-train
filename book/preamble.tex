% ============================================================
%  Trening Mózgu -- book preamble
%
%  THE UNIT IS A SET, NOT AN EXERCISE.
%
%  A set is a run of exercises listed one under another, done against a
%  stopwatch and scored as a whole. Answers live in one appendix at the BACK of
%  the book, never beside the exercise. How many exercises that is belongs to
%  tools/gen_sets.py, which holds it in one constant: two sets a reader times
%  against each other have to be the same length or the two numbers mean
%  nothing.
%
%  This is a deliberate replacement of an earlier design that gave each
%  exercise its own frame, with the answer opening the next one. That design is
%  right for programmed instruction, where the point is to LEARN a step. It is
%  wrong here: this book drills, and what it measures is how many you get
%  through and how fast. One exercise per frame put two on a page, made the
%  page ceremony rather than work, and put the answer where it interrupts the
%  run. Nothing about it is kept.
%
%  Engine: pdfLaTeX. Do NOT load newtx -- the companion repository probes for
%  it and degrades, which made its TS1 font map a trap on a half-installed
%  machine. This preamble simply never asks.
% ============================================================

% ------------------------------------------------------------
%  The answer store
% ------------------------------------------------------------
% Answers are written at the point of use and printed at the BACK. Carrying
% them there through a global macro store rather than an .aux file is
% deliberate: material written to an aux-style file is expanded at shipout,
% where a fragile command breaks with an error naming neither the answer nor
% the set it came from. \include is sequential within a run, so a macro defined
% globally while typesetting set 3 is still defined when the appendix is set.
%
% \csxappto expands \thezad at store time -- the number has to be frozen when
% the exercise is written, not when the appendix is -- while \unexpanded keeps
% the answer's own tokens literal, so $ and \frac survive to the back page.
\newcounter{zestaw}
\newcounter{zad}[zestaw]

% \btsep, not \quad. An answer list is a long run of short unbreakable items
% with no hyphenation anywhere in it, so a rigid separator leaves TeX no legal
% breakpoint it can afford and every line comes out overfull -- measured at
% 12.0pt, seventeen times, on the first build of this appendix. The penalty
% offers the breakpoint and the stretch lets the line end early.
\newcommand{\btsep}{\penalty0\hskip 0.9em plus 0.6em minus 0.25em\relax}
\newcommand{\btstoreans}[1]{%
  \csxappto{btans@\thezestaw}{\thezad)~\unexpanded{#1}\noexpand\btsep}%
}

\newlength{\btzadskip}
\setlength{\btzadskip}{3.2pt}

% ------------------------------------------------------------
%  \z -- one exercise on one line
% ------------------------------------------------------------
% A number, the problem, and a rule to write on. That is the whole element.
% There is no difficulty star and no per-exercise timer: both are properties of
% the SET, printed once at its head, and repeating them on every line would be
% noise that costs exercises.
%
% \strut so rows keep an even pitch whether or not the problem has a fraction
% or an exponent in it -- a column of drill lines that breathes unevenly is
% harder to run down with a pencil.
\newcommand{\z}[2]{%
  \stepcounter{zad}%
  \btstoreans{#2}%
  \par\noindent
  \makebox[2.1em][r]{\color{btgrey}\footnotesize\thezad.\ }%
  #1\,\hfill\rule[-0.2ex]{2.6em}{0.4pt}%
  \strut\par\vspace{\btzadskip}%
}

% Wider exercises -- a word puzzle, a sequence with many terms -- that will not
% sit three to a row. Same numbering and the same answer store; only the
% surrounding column count differs, and that is the set's business.
\newcommand{\zz}[2]{%
  \stepcounter{zad}%
  \btstoreans{#2}%
  \par\noindent
  \makebox[2.1em][r]{\color{btgrey}\footnotesize\thezad.\ }%
  % The parbox must leave room for EVERYTHING beside it: the 2.1em number box
  % before it and the 2.6em answer rule after it, plus a gap. Reserving 3.6em
  % for 4.7em of furniture overflowed every wide line by exactly the missing
  % 1.1em -- 12.045pt at 11pt, and the constant width across sixteen boxes is
  % what said it was arithmetic rather than bad line breaking.
  \parbox[t]{\dimexpr\linewidth-5.6em\relax}{\raggedright #1\strut}%
  \hfill\rule[-0.2ex]{2.6em}{0.4pt}%
  \par\vspace{\btzadskip}%
}

% ------------------------------------------------------------
%  Columns, and why they are not multicol's
% ------------------------------------------------------------
% A set is laid out as one row of minipages, and the generator says where the
% column breaks fall. multicol did this until the sets grew to forty, and it
% cannot do the one thing this format needs: A SET MUST NOT BE SPLIT.
%
% multicols cannot be put in a box -- it says so and refuses -- so a set built
% with it is always breakable, and at forty exercises the breaks started
% landing between the last exercise and the Czas/Poprawne/Data box. A scoring
% box on the far side of a page turn from the run it scores is the one thing
% this book cannot ship: that box is the product.
%
% A row of minipages is a single horizontal box. It cannot be broken, so the
% set moves to the next page whole or not at all, and the \nobreak either side
% keeps the head and the foot with it. Two other things come free: the split is
% balanced exactly (14/13/13, where multicol gave 14/14/12 under
% \raggedcolumns), and a set too tall for a page is now an overfull \vbox the
% gate reports rather than a silent split nobody sees.
\newlength{\btcolsep}
\setlength{\btcolsep}{1.2em}
\newlength{\btcolwidth}
\newlength{\btcoltmp}

% \btnc -- column break. Emitted by tools/gen_sets.py between the counts
% split() computes; written by hand in the sets under book/sets/.
% \end followed by \begin of the SAME environment inside one macro is balanced
% as far as LaTeX's environment checking is concerned.
\newcommand{\btnc}{%
  \end{minipage}\hspace{\btcolsep}\begin{minipage}[t]{\btcolwidth}%
}

% Set \btcolwidth for #1 columns. Done with \multiply and \divide on the
% registers rather than one \dimexpr: the expression needs a nested \numexpr
% for #1-1 and reads worse than the four lines it saves.
\newcommand{\btsetcolwidth}[1]{%
  \setlength{\btcolwidth}{\btcolsep}%
  \multiply\btcolwidth by \numexpr#1-1\relax
  \setlength{\btcoltmp}{\linewidth}%
  \addtolength{\btcoltmp}{-\btcolwidth}%
  \setlength{\btcolwidth}{\btcoltmp}%
  \divide\btcolwidth by #1\relax
}

% ------------------------------------------------------------
%  The set
% ------------------------------------------------------------
% \begin{zestaw}{title}{stars}{target time}{columns}
%
% The head carries the three things that are true of every exercise below it --
% what kind, how hard, how long the whole set should take -- and the foot
% carries the two the reader writes in. Those two boxes are the point of the
% book: a set you did not time and did not score is just a page of sums.
\newcommand{\DifficultyStars}[1]{%
  \ifcase#1
  \or {\color{orange}\faStar}\faStar[regular]\faStar[regular]%
  \or {\color{orange}\faStar\faStar}\faStar[regular]%
  \or {\color{orange}\faStar\faStar\faStar}%
  \fi
}

% EVERY \vspace HERE IS PRECEDED BY \nobreak, and that is the whole point.
% `\par\vspace{4pt}` puts glue straight after a line box, which is a legal
% breakpoint, and the page-breaker took it: the Czas/Poprawne/Data line came
% out at the top of the next page with the rule that belongs above it left
% behind on the previous one. A penalty before the glue is what makes the box
% one piece -- the grid above it is already a single unbreakable box, so with
% this the whole set moves as a unit.
%
% #1 IS HOW MANY TIMES THE SET WILL BE SCORED, AND THE DEFAULT IS TWO.
%
% It was one, and that was a hole in the design rather than a saving. The book
% has always told the reader to come back to a set after a week and compare --
% it is the reason there is a date box at all -- and then gave them one line to
% write on. The plan makes the hole acute: it schedules that repeat on a named
% day for seventy-two of the sets, and the second measurement had nowhere to go
% except a page it could not be compared against.
%
% The benchmark sets take five, because they are done weekly for a month.
%
% The rules line up across rows without a tabular: every row is the same three
% fields separated by \hfill in a box of the same width, so the columns are a
% consequence of the geometry rather than something to declare.
\newcounter{btrow}
% THE GLUE AND ITS \nobreak GO BEFORE THE LINE, NEVER AFTER IT.
%
% After it, the last row of every foot ends with \nobreak, the next set's head
% opens with one, and the chain runs the length of the book: TeX is left with
% no legal breakpoint anywhere and sets the whole thing as one column. Measured,
% because it was written the other way round first -- 30 pages instead of 86,
% and eight overfull vboxes the tallest of which was seven metres.
\newcommand{\btfootline}[1]{%
  \ifnum#1>1\relax\par\nobreak\vspace{2.5pt}\fi
  \par\noindent
  % The split check's label rides on the FIRST row, in horizontal mode. Left in
  % vertical mode it records the page TeX is on, which is the page the previous
  % material went to -- so a foot that lands overleaf takes its own head's page
  % number, the check compares a page against itself, and a set ships with its
  % scoring box on the far side of a page turn. That is not hypothetical: this
  % refactor did exactly that, with a green build, and the render found it.
  \ifnum#1=1\relax\label{btsetfoot\thezestaw}\fi
  {\color{btgrey}\footnotesize#1.}\ \ {\bfseries Czas:}~\rule[-0.4ex]{6em}{0.4pt}\hfill
  {\bfseries Poprawne:}~\rule[-0.4ex]{3em}{0.4pt}\,/\,\thezad\hfill
  {\bfseries Data:}~\rule[-0.4ex]{6em}{0.4pt}%
  \par
}

% The lines are accumulated BEFORE anything is typeset. A \loop that emits
% material directly is fine here, but the same pattern is needed the moment
% this goes back inside an alignment -- a \loop in a tabular body does not
% survive scanning, because TeX needs to see the & and the row break while it
% reads the cell and a conditional hides them. The symptom is an empty table
% and no error at all.
\newcommand{\btfootbody}{}
\newcommand{\btmakefootbody}[1]{%
  \renewcommand{\btfootbody}{}%
  \setcounter{btrow}{0}%
  \loop\ifnum\value{btrow}<#1\relax
    \stepcounter{btrow}%
    \eappto\btfootbody{\noexpand\btfootline{\thebtrow}}%
  \repeat
}

\newcommand{\btsetfoot}[1]{%
  \par\nobreak\vspace{6pt}%
  \noindent\textcolor{btrule}{\rule{\linewidth}{0.5pt}}\par\nobreak\vspace{4pt}%
  \btmakefootbody{#1}%
  \btfootbody
}

% A SET THAT IS SPLIT IS A BUILD FAILURE, not something to catch by eye.
%
% The layout above is built so a set moves whole, but "built so it cannot
% happen" is what was believed about the version that shipped a scoring box on
% the wrong side of a page turn. This measures it instead: the head and the
% foot each carry a label, and if the two pages differ the run stops with an
% error that tools/checklog.py counts.
%
% \getpagerefnumber gives 0 for a label that does not exist yet, which is every
% label on the first pass -- the comparison waits for a pass that has an .aux.
\newcommand{\btwholecheck}{%
  \edef\btheadpg{\getpagerefnumber{btsethead\thezestaw}}%
  \edef\btfootpg{\getpagerefnumber{btsetfoot\thezestaw}}%
  \ifnum\btheadpg>0
    \ifnum\btheadpg=\btfootpg\else
      \PackageError{trening-mozgu}{%
        Zestaw \thezestaw\space is split: head on page \btheadpg,
        foot on page \btfootpg}{%
        A set is timed and scored as one run, so it has to sit on one page.
        Shorten it, give it more columns, or make the page taller}%
    \fi
  \fi
}

% \begin{zestaw}[rows]{title}{stars}{target}{cols}
%
% The optional argument is the number of scoring rows. TWO by default, because
% the plan schedules a repeat of nearly every set and the reader needs both
% times in one place; five for the benchmark sets, done weekly for a month.
% THE NUMBER A SET COMES UP UNDER IS PART OF THE BOOK'S CONTRACT WITH THE PLAN.
%
% The plan prints `Zestaw 41` as a literal, computed by tools/gen_plan.py from
% the generator's own lists. The number the reader sees comes from a counter
% stepping through structure.tex. Those are the same thing worked out twice,
% and nothing made them agree: reorder a chapter and every row of the plan
% points one set off, silently, with every other gate green.
%
% So the generated include lists emit \btexpect{n} before each \input, and a
% set that comes up under any other number stops the build. A set reached with
% no expectation at all is an error too -- otherwise an \input added straight
% to structure.tex would slip past, which is exactly the mistake this guards.
\newcommand{\btexpectnum}{}
\newcommand{\btexpect}[1]{\gdef\btexpectnum{#1}}

\newenvironment{zestaw}[5][2]{%
  \refstepcounter{zestaw}%
  \ifx\btexpectnum\empty
    \PackageError{trening-mozgu}{%
      Zestaw \thezestaw\space has no \string\btexpect}{%
      Every set is included from a generated list in sets/generated/, which
      emits its number. Add the set to tools/gen_sets.py and run `make sets`
      rather than putting an \string\input in structure.tex}%
  \else
    \ifnum\btexpectnum=\thezestaw\relax\else
      \PackageError{trening-mozgu}{%
        Zestaw \thezestaw\space expected number \btexpectnum}{%
        The plan names sets by number, so the book's order and
        tools/gen_sets.py's book_order() have to agree. Run `make sets` and
        `make plan`, or put the chapter back where it was}%
    \fi
  \fi
  \gdef\btexpectnum{}%
  \par\addvspace{10pt}%
  % Head
  \noindent\textcolor{btink}{\rule{\linewidth}{1.2pt}}\par\nobreak\vspace{3pt}%
  \noindent
  {\large\bfseries ZESTAW \thezestaw}\label{btsethead\thezestaw}\quad{\large #2}%
  \hfill \DifficultyStars{#3}\quad
  \tikz[baseline=(b.base)]{\node[fill=black!8, rounded corners=3pt,
    inner xsep=5pt, inner ysep=2.5pt, outer sep=0pt,
    font=\small\bfseries] (b) {\faClock[regular]\ cel: #4};}%
  \par\nobreak\vspace{2pt}%
  \noindent\textcolor{btrule}{\rule{\linewidth}{0.5pt}}\par\nobreak\vspace{7pt}%
  \addcontentsline{toc}{section}{Zestaw \thezestaw\ --- #2}%
  % The row count is read at \end, where the argument is long gone, so it is
  % stashed here. \gdef because the environment's two halves are a group apart.
  \gdef\btsetrows{#1}%
  \btsetcolwidth{#5}%
  \nobreak
  \noindent\begin{minipage}[t]{\btcolwidth}%
}{%
  \end{minipage}%
  \par\nobreak
  \btsetfoot{\btsetrows}%
  \btwholecheck
}

% ------------------------------------------------------------
%  The answers appendix
% ------------------------------------------------------------
% Every set's answers, in order, at the back. \ifcsdef rather than an assumed
% range: a set that stored nothing prints nothing, instead of an empty heading
% that reads like a missing page.
% THE RUNNING HEAD CARRIES THE RANGE OF SETS ON THE PAGE.
%
% The appendix is a lookup table and it is now eight pages of eleven sets each.
% A head reading `Odpowiedzi` on all eight tells the reader nothing they did
% not know; `Odpowiedzi 26--36` is the thing a lookup table's head is for.
%
% \rightmark is the FIRST mark on the page and \leftmark the LAST, which is
% what makes a range out of one \markboth per set. Both are set to the same
% number because the two arguments are being used as first-and-last here
% rather than as chapter-and-section.
\fancypagestyle{btodp}{%
  \fancyhf{}%
  \fancyhead[LE,RO]{\small\thepage}%
  \fancyhead[RE,LO]{\small\itshape Odpowiedzi\quad\rightmark--\leftmark}%
  \renewcommand{\headrulewidth}{0.4pt}%
}

\newcounter{btansi}
\newcommand{\odpowiedzi}{%
  \chapter*{Odpowiedzi}%
  \addcontentsline{toc}{chapter}{Odpowiedzi}%
  \markboth{}{}%
  \pagestyle{btodp}%
  \small\raggedright
  \setlength{\parindent}{0pt}%
  \setcounter{btansi}{0}%
  \loop\ifnum\value{btansi}<\value{zestaw}%
    \stepcounter{btansi}%
    \ifcsdef{btans@\thebtansi}{%
      % One block per set, in a box so it cannot split. A page that begins in
      % the middle of a list begins with numbers and no heading, and the
      % appendix is a lookup table -- a row you cannot see the label of is a
      % row you have to leave the page to identify. A block is a few lines
      % long, so keeping it whole costs nothing.
      \par\addvspace{7pt}%
      \noindent\begin{minipage}{\linewidth}%
        {\bfseries\color{btink}Zestaw \thebtansi}\par\nobreak\vspace{1pt}%
        \csuse{btans@\thebtansi}\par
      % THE MARK GOES AFTER THE BLOCK, AND NOT INSIDE IT.
      %
      % Inside it is worse than wrong, it is silent: a \markboth in a minipage
      % never reaches the page at all, so the head came out `Odpowiedzi --`
      % with both ends empty and no warning anywhere.
      %
      % After the block it is also exactly right, which before it would not
      % have been. A block is unbreakable, so the first mark on a page is the
      % first block that ENDS there, which is the first block on it, and the
      % last mark is the last block on it. Placed before the block instead,
      % a mark whose block is pushed overleaf stays behind and the head claims
      % a set the page does not carry.
      \end{minipage}\markboth{\thebtansi}{\thebtansi}\par
    }{}%
  \repeat
}

% ------------------------------------------------------------
%  The progress grid
% ------------------------------------------------------------
% One per benchmark set. Five numbers in a column is a table; five points on a
% grid is a trend, and the difference is the whole reason the benchmark sets
% exist at all.
%
% THE Y AXIS CARRIES NO NUMBERS, and that is not laziness. One reader's first
% measurement is four minutes and another's is eleven, so any printed scale is
% wrong for both -- and a scale chosen by the book would also choose how large
% an improvement looks. The reader writes their own against the rules on the
% left, which scales the axis to their range.
\newcommand{\btsiatka}[2]{%
  \begin{minipage}[t]{0.47\linewidth}%
    {\bfseries\color{btink}#2}\hfill{\small\color{btgrey}Zestaw #1}%
    \par\vspace{4pt}%
    \begin{tikzpicture}[x=1.05cm, y=0.5cm]
      \draw[btrule, very thin] (0,0) grid (5,8);
      \draw[btink, thick] (0,0) -- (0,8);
      \draw[btink, thick] (0,0) -- (5,0);
      \foreach \i in {1,...,5}
        \node[below, font=\footnotesize\color{btgrey}] at (\i-0.5, 0) {\i};
      \foreach \j in {1,...,8}
        \node[left, inner xsep=2pt] at (0,\j) {\rule{1.4em}{0.3pt}};
    \end{tikzpicture}%
    \par\vspace{2pt}%
    {\footnotesize\color{btgrey}pomiar\hfill czas w dół to postęp}%
  \end{minipage}%
}
